cmake_minimum_required(VERSION 3.16)
project(fp32_mode2_transfer_holdout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(panel_ab_kernel STATIC panel_ab_kernel.cpp)
add_executable(panel_ab_probe panel_ab_runner.cpp)
target_include_directories(panel_ab_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(panel_ab_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(panel_ab_probe PRIVATE panel_ab_kernel ascendcl)
target_link_options(panel_ab_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(panel_ab_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
