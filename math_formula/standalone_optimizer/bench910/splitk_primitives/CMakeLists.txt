cmake_minimum_required(VERSION 3.16)
project(splitk_primitives LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "CANN build type" FORCE)
endif()
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
set(MATMUL_KERNEL_DIR "" CACHE PATH "mat_mul_v3/op_kernel source directory")
if(NOT ASCEND_CANN_PACKAGE_PATH OR NOT MATMUL_KERNEL_DIR)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH and MATMUL_KERNEL_DIR")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
configure_file(probe_kernel.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/probe_kernel.cpp @ONLY)
ascendc_library(splitk_probe_kernel STATIC ${CMAKE_CURRENT_BINARY_DIR}/probe_kernel.cpp)
add_executable(splitk_probe probe_runner.cpp)
target_include_directories(splitk_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(splitk_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(splitk_probe PRIVATE splitk_probe_kernel ascendcl)
target_link_options(splitk_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(splitk_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
