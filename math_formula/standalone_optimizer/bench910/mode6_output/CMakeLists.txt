cmake_minimum_required(VERSION 3.16)
project(mode6_output_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(mode6_output_kernel STATIC probe_kernel.cpp)
add_executable(mode6_output_probe probe_runner.cpp)
target_include_directories(mode6_output_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(mode6_output_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(mode6_output_probe PRIVATE mode6_output_kernel ascendcl)
target_link_options(mode6_output_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(mode6_output_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
