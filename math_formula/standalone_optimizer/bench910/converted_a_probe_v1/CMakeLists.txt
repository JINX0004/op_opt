cmake_minimum_required(VERSION 3.16)
project(mmv3_converted_a_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
set(MMV3_KERNEL_SOURCE "" CACHE PATH "MatMulV3 op_kernel directory")
if(NOT ASCEND_CANN_PACKAGE_PATH OR NOT MMV3_KERNEL_SOURCE)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH and MMV3_KERNEL_SOURCE")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(mmv3_converted_a_kernel STATIC probe_kernel.cpp)
target_include_directories(mmv3_converted_a_kernel PRIVATE ${MMV3_KERNEL_SOURCE})
add_executable(mmv3_converted_a_probe probe_runner.cpp)
target_include_directories(mmv3_converted_a_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(mmv3_converted_a_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(mmv3_converted_a_probe PRIVATE mmv3_converted_a_kernel ascendcl)
target_link_options(mmv3_converted_a_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(mmv3_converted_a_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
