cmake_minimum_required(VERSION 3.16)
project(fp32_base_source_stream LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(base_stream_probe_kernel STATIC probe_kernel.cpp)
add_executable(base_stream_probe probe_runner.cpp)
target_include_directories(base_stream_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(base_stream_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(base_stream_probe PRIVATE base_stream_probe_kernel ascendcl)
target_link_options(base_stream_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(base_stream_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
