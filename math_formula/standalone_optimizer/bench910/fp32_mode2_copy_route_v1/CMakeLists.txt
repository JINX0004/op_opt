cmake_minimum_required(VERSION 3.16)
project(fp32_mode2_copy_route LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(copy_route_kernel STATIC copy_route_kernel.cpp)
add_executable(copy_route_probe copy_route_runner.cpp)
target_include_directories(copy_route_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(copy_route_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(copy_route_probe PRIVATE copy_route_kernel ascendcl)
target_link_options(copy_route_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(copy_route_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
ascendc_library(ab_route_kernel STATIC ab_route_kernel.cpp)
add_executable(ab_route_probe ab_route_runner.cpp)
target_include_directories(ab_route_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(ab_route_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(ab_route_probe PRIVATE ab_route_kernel ascendcl)
target_link_options(ab_route_probe PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(ab_route_probe PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
