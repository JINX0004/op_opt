cmake_minimum_required(VERSION 3.16)
project(fp32_mode2_stage_ladder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B3" CACHE STRING "target SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "installed CANN root")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(probe_kernel STATIC probe_kernel.cpp)
add_executable(mode2_stage_ladder probe_runner.cpp)
target_include_directories(mode2_stage_ladder PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(mode2_stage_ladder PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(mode2_stage_ladder PRIVATE probe_kernel ascendcl)
target_link_options(mode2_stage_ladder PRIVATE "LINKER:--allow-shlib-undefined")
set_target_properties(mode2_stage_ladder PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
